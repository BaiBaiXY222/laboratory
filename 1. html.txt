        import * as THREE from 'three';

        const controls = new OrbitControls(camera, renderer.domElement);
        controls.target.set(0, 5.5, 0);
        controls.enableDamping = true;
        controls.dampingFactor = 0.08;
        controls.minDistance = 8;
        controls.maxDistance = 40;
        controls.maxPolarAngle = Math.PI * 0.7;

        // ==================== 光照 ====================

        const keyLight = new THREE.DirectionalLight(0xffffff, 6);

        keyLight.shadow.camera.left = -20;

        ground.receiveShadow = true;

        // 网格线
        const grid = new THREE.GridHelper(40, 40, 0x444466, 0x222233);

        // ==================== 实验台 ====================
        const tableGeo = new THREE.CylinderGeometry(2.5, 2.8, 0.6, 32);

        // ==================== 焦利秤支架 ====================
        const standGroup = new THREE.Group();
        
        // 底座
        const baseGeo = new THREE.CylinderGeometry(1.2, 1.4, 0.3, 32);
        const baseMat = new THREE.MeshStandardMaterial({ color: 0x37474F, roughness: 0.3, metalness: 0.7 });

        // 竖杆

        const pole = new THREE.Mesh(poleGeo, poleMat);

        standGroup.add(pole);
        
        // 顶部横梁
        const beamGeo = new THREE.BoxGeometry(0.35, 0.25, 1.8);
        const beamMat = new THREE.MeshStandardMaterial({ color: 0x546E7A, roughness: 0.3, metalness: 0.8 });
        const beam = new THREE.Mesh(beamGeo, beamMat);
        beam.position.set(0, 14.1, 0.3);

        standGroup.add(beam);
        
        // 顶部挂钩
        const hookGeo = new THREE.TorusGeometry(0.2, 0.06, 8, 16, Math.PI);
        const hookMat = new THREE.MeshStandardMaterial({ color: 0xB0BEC5, roughness: 0.2, metalness: 0.9 });
        const hook = new THREE.Mesh(hookGeo, hookMat);
        hook.position.set(0, 13.85, 0.65);
        hook.rotation.x = Math.PI/2;
        hook.rotation.z = Math.PI;
        hook.castShadow = true;
        standGroup.add(hook);
        
        standGroup.position.set(0, 0, 0);
        scene.add(standGroup);

        // ==================== 弹簧（动态） ====================
        const springGroup = new THREE.Group();
        const springPivot = new THREE.Group();
        springPivot.position.set(0, 13.75, 0.65);
        springGroup.add(springPivot);
        
        let springLength = 4.0; // 当前弹簧长度
        const springRestLength = 4.0; // 原长
        const springCoils = 30;
        const springRadius = 0.25;
        
        function createSpring(length) {
            // 清除旧弹簧
            while(springPivot.children.length > 0) {
                springPivot.remove(springPivot.children[0]);
            }
            
            const points = [];
            const numPoints = springCoils * 8;
            for (let i = 0; i <= numPoints; i++) {
                const t = i / numPoints;
                const y = -t * length;
                const angle = t * springCoils * Math.PI * 2;
                const x = Math.cos(angle) * springRadius;
                const z = Math.sin(angle) * springRadius;
                points.push(new THREE.Vector3(x, y, z));
            }
            
            const curve = new THREE.CatmullRomCurve3(points);
            const tubeGeo = new THREE.TubeGeometry(curve, 128, 0.04, 8, false);
            const tubeMat = new THREE.MeshStandardMaterial({ 
                color: 0x90A4AE, roughness: 0.2, metalness: 0.9 
            });
            const springMesh = new THREE.Mesh(tubeGeo, tubeMat);
            springMesh.castShadow = true;
            springMesh.name = "spring";
            springPivot.add(springMesh);
            
            // 弹簧底部连接点
            const bottomDotGeo = new THREE.SphereGeometry(0.12, 16, 16);
            const bottomDotMat = new THREE.MeshStandardMaterial({ color: 0xB0BEC5, roughness: 0.2, metalness: 0.9 });
            const bottomDot = new THREE.Mesh(bottomDotGeo, bottomDotMat);
            bottomDot.position.set(0, -length, 0);
            bottomDot.name = "bottomDot";
            springPivot.add(bottomDot);
        }
        
        createSpring(springLength);
        scene.add(springGroup);

        // ==================== 金属环挂钩与环 ====================
        const ringGroup = new THREE.Group();
        
        // 连接细杆
        const rodGeo = new THREE.CylinderGeometry(0.04, 0.04, 2.5, 16);
        const rodMat = new THREE.MeshStandardMaterial({ color: 0xB0BEC5, roughness: 0.2, metalness: 0.9 });
        const rod = new THREE.Mesh(rodGeo, rodMat);
        rod.position.y = -1.25;
        rod.castShadow = true;
        ringGroup.add(rod);
        
        // 金属环（外环 + 内环）
        function createRing(radius, yOffset, color) {
            const torusGeo = new THREE.TorusGeometry(radius, 0.06, 16, 64);
            const torusMat = new THREE.MeshStandardMaterial({ color, roughness: 0.2, metalness: 0.95 });
            const torus = new THREE.Mesh(torusGeo, torusMat);
            torus.rotation.x = Math.PI/2;
            torus.position.y = yOffset;
            torus.castShadow = true;
            return torus;
        }
        
        // 外径3.50cm → 半径1.75cm = 0.0175m，场景缩放：1m=1unit, 所以用0.175（放大10倍便于观察）
        const scaleFactor = 10; // 将cm放大10倍便于观察
        const outerRadius = 1.75 * scaleFactor / 100; // 0.175 units
        const innerRadius = 1.65 * scaleFactor / 100; // 0.165 units
        
        const outerRing = createRing(outerRadius, -3.8, 0xCFD8DC);
        const innerRing = createRing(innerRadius, -3.8, 0xB0BEC5);
        
        ringGroup.add(outerRing);
        ringGroup.add(innerRing);
        
        // 连接内外环的细丝（4根）
        for (let i = 0; i < 4; i++) {
            const angle = (i / 4) * Math.PI * 2;
            const connGeo = new THREE.CylinderGeometry(0.02, 0.02, outerRadius - innerRadius, 8);
            const conn = new THREE.Mesh(connGeo, rodMat);
            conn.position.set(
                Math.cos(angle) * (outerRadius + innerRadius) / 2,
                -3.8,
                Math.sin(angle) * (outerRadius + innerRadius) / 2
            );
            conn.rotation.z = Math.PI/2;
            conn.rotation.y = -angle;
            conn.castShadow = true;
            ringGroup.add(conn);
        }
        
        ringGroup.position.set(0, 9.75 - springLength, 0.65);
        scene.add(ringGroup);

        // ==================== 玻璃皿与液面 ====================
        const dishGroup = new THREE.Group();
        
        // 玻璃皿
        const dishOuterGeo = new THREE.CylinderGeometry(2.5, 2.2, 1.5, 48, 1, true);
        const glassMat = new THREE.MeshPhysicalMaterial({
            color: 0xcfecff, roughness: 0.05, metalness: 0.05,
            transparent: true, opacity: 0.35, clearcoat: 0.3,
            envMapIntensity: 0.5
        });
        const dishOuter = new THREE.Mesh(dishOuterGeo, glassMat);
        dishOuter.position.y = 0.75;
        dishOuter.castShadow = true;
        dishGroup.add(dishOuter);
        
        const dishBottomGeo = new THREE.CylinderGeometry(2.2, 2.2, 0.1, 48);
        const dishBottom = new THREE.Mesh(dishBottomGeo, glassMat);
        dishBottom.position.y = 0.05;
        dishGroup.add(dishBottom);
        
        // 液体
        const liquidGeo = new THREE.CylinderGeometry(2.15, 2.15, 1.1, 48);
        const liquidMat = new THREE.MeshPhysicalMaterial({
            color: 0x4488cc, roughness: 0.1, metalness: 0.0,
            transparent: true, opacity: 0.55, clearcoat: 0.5
        });
        const liquid = new THREE.Mesh(liquidGeo, liquidMat);
        liquid.position.y = 0.6;
        liquid.name = "liquid";
        dishGroup.add(liquid);
        
        // 液面（用于视觉参考）
        const surfaceGeo = new THREE.CylinderGeometry(2.14, 2.14, 0.02, 48);
        const surfaceMat = new THREE.MeshPhysicalMaterial({
            color: 0x88bbff, roughness: 0.02, metalness: 0.1,
            transparent: true, opacity: 0.7, clearcoat: 1.0
        });
        const surface = new THREE.Mesh(surfaceGeo, surfaceMat);
        surface.position.y = 1.15;
        surface.name = "surface";
        dishGroup.add(surface);
        
        dishGroup.position.set(0, 0.3, 0);
        scene.add(dishGroup);

        // ==================== 刻度尺 ====================
        const rulerGroup = new THREE.Group();
        
        const rulerBackGeo = new THREE.BoxGeometry(0.6, 8, 0.08);
        const rulerBackMat = new THREE.MeshStandardMaterial({ color: 0xFFFDE7, roughness: 0.5 });
        const rulerBack = new THREE.Mesh(rulerBackGeo, rulerBackMat);
        rulerBack.position.y = 4;
        rulerGroup.add(rulerBack);
        
        // 刻度线
        for (let i = 0; i <= 40; i++) {
            const tickLength = i % 5 === 0 ? 0.25 : (i % 10 === 0 ? 0.35 : 0.15);
            const tickGeo = new THREE.BoxGeometry(tickLength, 0.02, 0.02);
            const tick = new THREE.Mesh(tickGeo, new THREE.MeshStandardMaterial({ color: 0x333333 }));
            tick.position.set(0.15, i * 0.2, 0.05);
            rulerGroup.add(tick);
        }
        
        rulerGroup.position.set(1.5, 3.5, 1.2);
        rulerGroup.rotation.y = -0.3;
        scene.add(rulerGroup);

        // ==================== 粒子效果（液面张力） ====================
        const particlesGroup = new THREE.Group();
        const particlesGeo = new THREE.BufferGeometry();
        const particlesCount = 200;
        const particlesPositions = new Float32Array(particlesCount * 3);
        for (let i = 0; i < particlesCount; i++) {
            const angle = Math.random() * Math.PI * 2;
            const r = Math.random() * outerRadius * 1.5;
            particlesPositions[i * 3] = Math.cos(angle) * r;
            particlesPositions[i * 3 + 1] = -3.78;
            particlesPositions[i * 3 + 2] = Math.sin(angle) * r;
        }
        particlesGeo.setAttribute('position', new THREE.BufferAttribute(particlesPositions, 3));
        const particlesMat = new THREE.PointsMaterial({ 
            color: 0xaaddff, size: 0.015, transparent: true, opacity: 0.6,
            blending: THREE.AdditiveBlending
        });
        const particles = new THREE.Points(particlesGeo, particlesMat);
        particles.visible = false;
        particlesGroup.add(particles);
        ringGroup.add(particlesGroup);

        // ==================== 状态变量 ====================
        const k = 0.352; // N/m
        const D1 = 3.50e-2; // m
        const D2 = 3.30e-2; // m
        const L = Math.PI * (D1 + D2); // 有效周长
        let currentDisplacement = 0; // 弹簧伸长量(mm)
        let isDetached = false;
        let platformOffset = 0;
        
        // 动画参数
        let targetSpringLength = springRestLength;
        let animating = false;

        // ==================== UI更新 ====================
        function updateUI() {
            const force = k * (currentDisplacement / 1000); // N
            const alpha = isDetached ? (force / L) : null;
            
            document.getElementById('scaleVal').textContent = currentDisplacement.toFixed(2);
            document.getElementById('forceVal').textContent = force.toFixed(4);
            if (alpha !== null && currentDisplacement > 0.5) {
                document.getElementById('alphaVal').textContent = (alpha * 1000).toFixed(2);
            } else if (isDetached) {
                document.getElementById('alphaVal').textContent = '--';
            } else {
                document.getElementById('alphaVal').textContent = '--';
            }
        }

        // ==================== 同步弹簧与环位置 ====================
        function syncSpringAndRing() {
            const actualLength = springRestLength + currentDisplacement / 100;
            createSpring(actualLength);
            ringGroup.position.y = springPivot.position.y - actualLength;
        }

        // ==================== 按钮事件 ====================
        document.getElementById('btnDown').addEventListener('click', () => {
            if (animating) return;
            if (!isDetached) {
                currentDisplacement = Math.max(0, currentDisplacement - 1.0);
            }
            syncSpringAndRing();
            updateUI();
        });
        
        document.getElementById('btnUp').addEventListener('click', () => {
            if (animating) return;
            if (!isDetached) {
                currentDisplacement += 1.0;
            }
            syncSpringAndRing();
            updateUI();
        });
        
        document.getElementById('btnDetach').addEventListener('click', () => {
            if (animating || isDetached) return;
            animating = true;
            
            // 动画：环快速上升模拟拉脱
            const detachDisplacement = currentDisplacement + 3.0;
            const startVal = currentDisplacement;
            const duration = 800;
            const startTime = Date.now();
            
            function animateDetach() {
                const elapsed = Date.now() - startTime;
                const progress = Math.min(elapsed / duration, 1.0);
                // ease-out
                const eased = 1 - Math.pow(1 - progress, 3);
                currentDisplacement = startVal + (detachDisplacement - startVal) * eased;
                
                syncSpringAndRing();
                updateUI();
                
                if (progress < 1) {
                    requestAnimationFrame(animateDetach);
                } else {
                    isDetached = true;
                    particles.visible = true;
                    animating = false;
                    // 粒子动画
                    animateParticles();
                    updateUI();
                }
            }
            
            animateDetach();
        });
        
        document.getElementById('btnReset').addEventListener('click', () => {
            currentDisplacement = 0;
            isDetached = false;
            particles.visible = false;
            animating = false;
            syncSpringAndRing();
            updateUI();
        });
        
        // 键盘控制
        window.addEventListener('keydown', (e) => {
            if (animating) return;
            switch(e.key.toLowerCase()) {
                case 'arrowup':
                case 'w':
                    if (!isDetached) currentDisplacement += 0.5;
                    break;
                case 'arrowdown':
                case 's':
                    if (!isDetached) currentDisplacement = Math.max(0, currentDisplacement - 0.5);
                    break;
                case 'd':
                    document.getElementById('btnDetach').click();
                    break;
                case 'r':
                    document.getElementById('btnReset').click();
                    break;
            }
            syncSpringAndRing();
            updateUI();
        });

        // ==================== 粒子动画 ====================
        function animateParticles() {
            if (!isDetached) {
                particles.visible = false;
                return;
            }
            particles.visible = true;
            const positions = particles.geometry.attributes.position.array;
            for (let i = 0; i < particlesCount; i++) {
                positions[i * 3 + 1] += (Math.random() - 0.5) * 0.02;
                if (positions[i * 3 + 1] < -4.5 || positions[i * 3 + 1] > -3.0) {
                    positions[i * 3 + 1] = -3.78;
                }
            }
            particles.geometry.attributes.position.needsUpdate = true;
            if (isDetached) requestAnimationFrame(animateParticles);
        }

        // ==================== 水滴效果（拉脱后） ====================
        const droplets = [];
        function createDroplet() {
            const size = 0.03 + Math.random() * 0.04;
            const dropletGeo = new THREE.SphereGeometry(size, 8, 8);
            const dropletMat = new THREE.MeshPhysicalMaterial({
                color: 0x5599dd, roughness: 0.0, metalness: 0.0,
                transparent: true, opacity: 0.8, clearcoat: 0.8
            });
            const droplet = new THREE.Mesh(dropletGeo, dropletMat);
            droplet.position.copy(ringGroup.position);
            droplet.position.y -= 3.5;
            droplet.position.x += (Math.random() - 0.5) * 0.1;
            droplet.position.z += (Math.random() - 0.5) * 0.1;
            droplet.userData = { velocity: -0.02 - Math.random() * 0.04, life: 1.0 };
            scene.add(droplet);
            droplets.push(droplet);
        }

        // ==================== 渲染循环 ====================
        function animate() {
            requestAnimationFrame(animate);
            
            controls.update();
            
            // 更新水滴
            if (isDetached && Math.random() < 0.3) {
                createDroplet();
            }
            for (let i = droplets.length - 1; i >= 0; i--) {
                const d = droplets[i];
                d.position.y += d.userData.velocity;
                d.userData.life -= 0.01;
                d.material.opacity = d.userData.life * 0.8;
                if (d.userData.life <= 0 || d.position.y < -3) {
                    scene.remove(d);
                    droplets.splice(i, 1);
                }
            }
            
            renderer.render(scene, camera);
        }

        // ==================== 窗口响应 ====================
        window.addEventListener('resize', () => {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        });

        // ==================== 启动 ====================
        syncSpringAndRing();
